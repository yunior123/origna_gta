123 |     function isValidOrderStateTransition(currentStatus, newStatus) {
124 |       // Define valid state transitions
125 |       // This prevents data corruption from invalid status changes
126 |       let validTransitions = {
127 |         'pending': ['confirmed', 'cancelled', 'failed', 'expired'], // Initial state -> completed states
128 |         'confirmed': ['processing', 'cancelled', 'expired'],       // After payment confirmed
129 |         'processing': ['shipped', 'cancelled'],                    // Seller processing
130 |         'shipped': ['in_transit', 'delivered'],                     // Shipped / in transit (cancel not allowed after ship)
131 |         'in_transit': ['delivered', 'cancelled'],                  // Carrier in transit
132 |         'delivered': ['refunded', 'partially_refunded', 'disputed'], // After delivery
133 |         'cancelled': [],                                            // Terminal state - no transitions
134 |         'failed': ['pending'],                                      // Retry -> pending
135 |         'expired': ['pending'],                                     // Retry -> pending
136 |         'refunded': [],                                             // Terminal state
137 |         'partially_refunded': ['refunded'],                         // Can become fully refunded
138 |         'disputed': ['refunded'],                                   // After dispute resolution
139 |       };
140 |       
141 |       // Get valid next states for current status
142 |       let allowedNextStates = validTransitions.get(currentStatus, []);
143 |       return newStatus in allowedNextStates;
144 |     }
